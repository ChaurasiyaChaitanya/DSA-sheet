class Solution {
    public long maxPairStrength(int[] nums) {
        int n = nums.length;
        long result = 0;

        for(int i=0; i<n; i++) {
            for(int j=i+1; j<n; j++) {
                long nu = ((long) nums[i] * (long) nums[j]);
                long de = gcd(nums[i], nums[j]);
                de *= de;

                result = Math.max(result, nu / de);
            }
        }

        return result;
    }

    private int gcd(int a, int b) {
        if(b == 0) return a; 
        return gcd(b, a % b); 
    }
}
