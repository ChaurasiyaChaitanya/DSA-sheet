class Solution {
    public int minMergedInteger(int[] nums) {
        int n = nums.length;
        int result = Integer.MAX_VALUE;

        for(int i=0; i<n; i+=2) {
            int temp = nums[i+1], multi = 1;
            while(temp > 0) {
                temp /= 10;
                multi *= 10;
            }
            result = Math.min(result, nums[i] * multi + nums[i+1]);
        }

        return result;
    }
}
