class Solution {
    public String reverseParentheses(String s) {
        int n = s.length();
        Stack<Integer> stack = new Stack<>();
        int[] pair = new int[n];

        for(int i=0; i<n; ++i) {
            if(s.charAt(i) == '(') {
                stack.push(i);
            }
            if(s.charAt(i) == ')') {
                int j = stack.pop();
                pair[i] = j;
                pair[j] = i;
            }
        }

        int dir = 1;
        StringBuilder sb = new StringBuilder();

        for(int i=0; i<n; i+=dir) {
            if(s.charAt(i) == '(' || s.charAt(i) == ')') {
                i = pair[i];
                dir *= -1;
            } else  {
                sb.append(s.charAt(i));
            }
        }

        return sb.toString();
    }
}
