class Solution {
    private static final int MOD = 1_000_000_007;

    public int numberOfSets(int n, int k) {
        int[] dp = new int[n];
        int[] prefixSum = new int[n+1];

        for(int i=0; i<n; i++) {
            dp[i] = 1;
            prefixSum[i+1] = (prefixSum[i] + dp[i]) % MOD;
        }

        for(int i=1; i<=k; i++) {
            dp[0] = 0;
            for(int j=1; j<n; j++) {
                dp[j] = (dp[j-1] + prefixSum[j]) % MOD;
            }
            for(int j=0; j<n; j++) {
                prefixSum[j+1] = (prefixSum[j] + dp[j]) % MOD;
            }
        }
        
        return dp[n-1];
    }
}
