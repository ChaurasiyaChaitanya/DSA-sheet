class Solution {
    public boolean uniformArray(int[] nums1) {
        int minEle = nums1[0];
        boolean hasOdd = false;

        for(int num : nums1) {
            if(num < minEle) {
                minEle = num;
            }
            if(num % 2 == 1) {
                hasOdd = true;
            }
        }

        if(minEle % 2 == 1) {
            return true;
        }

        return !hasOdd;
    }
}
