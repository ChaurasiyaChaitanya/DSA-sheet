class Solution {
    int n;
    String str;
    int[][] costDp;
    int[][] dp;

    public int minimumChanges(String s, int k) {
        n = s.length();
        str = s;

        costDp = new int[n][n];
        for(int[] row : costDp)
            Arrays.fill(row, Integer.MAX_VALUE);

        for(int i=0; i<n; i++) {
            for(int j=i+1; j<n; j++) {
                costDp[i][j] = getCost(i, j);
            }
        }

        dp = new int[n + 1][k + 1];
        for(int[] row : dp)
            Arrays.fill(row, Integer.MAX_VALUE);

        dp[n][0] = 0;

        for(int start=n; start>=0; start--) {
            for(int pieces=1; pieces<=k; pieces++) {
                if(start + 2 * pieces > n)
                    continue;

                int result = Integer.MAX_VALUE;

                for(int end=start+1; end<n; end++) {
                    int remainCost = dp[end + 1][pieces - 1];

                    if (remainCost == Integer.MAX_VALUE)
                        continue;

                    result = Math.min(result, costDp[start][end] + remainCost);
                }

                dp[start][pieces] = result;
            }
        }

        return dp[0][k];
    }

    private int getCost(int i, int j) {
        int len = j - i + 1;

        int minCost = Integer.MAX_VALUE;

        for(int d=1; d<len; d++) {
            if(len % d != 0)
                continue;

            int cost = 0;
            for(int group=0; group<d; group++) {
                int l = i + group;
                int r = l + (len / d - 1) * d;

                while(l < r) {
                    if(str.charAt(l) != str.charAt(r))
                        cost++;

                    l += d;
                    r -= d;
                }
            }

            minCost = Math.min(minCost, cost);
        }

        return minCost;
    }
}
