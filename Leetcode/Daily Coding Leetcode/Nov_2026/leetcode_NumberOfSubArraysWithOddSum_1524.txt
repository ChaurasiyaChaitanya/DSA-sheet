class Solution {
    private static final int MOD = 1_000_000_007;

    public int numOfSubarrays(int[] arr) {
        int oddCount = 0;
        int evenCount = 1; // prefix sum of 0 is even
        int result = 0, prefixSum = 0;

        for(int ele : arr) {
            prefixSum += ele;
            if((prefixSum & 1) == 1) {
                result = (result + evenCount) % MOD;
                oddCount++;
            } else {
                result = (result + oddCount) % MOD;
                evenCount++;
            }
        }

        return result;
    }
}
