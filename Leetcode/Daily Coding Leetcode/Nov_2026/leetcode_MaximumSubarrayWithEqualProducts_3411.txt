class Solution {
    public int maxLength(int[] nums) {
        int n = nums.length;
        int result = 0;

        for(int i=0; i<n; i++) {
            long prod = nums[i], gcd = nums[i], lcm = nums[i];
            for(int j=i+1; j<n; j++) {
                prod *= nums[j];
                gcd = gcd(gcd, nums[j]);
                lcm = lcm(lcm, nums[j]);

                if(prod == gcd * lcm) {
                    result = Math.max(result, j - i + 1);
                }
            }
        }

        return result;
    }

    private long gcd(long a, long b) {
        return b == 0 ? a : gcd(b, a % b);
    }

    private long lcm(long a, long b) {
        return a / gcd(a, b) * b;
    }
}
