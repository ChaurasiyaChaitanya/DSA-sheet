class Solution {
    public List<Integer> busiestServers(int k, int[] arrival, int[] load) {
        TreeSet<Integer> available = new TreeSet<>();
        for(int i=0; i<k; i++) {
            available.add(i);
        }

        PriorityQueue<int[]> busy = new PriorityQueue<>((a, b) ->
            a[0] != b[0] ? Integer.compare(a[0], b[0]) : Integer.compare(a[1], b[1])
        );

        int[] countReq = new int[k];
        for(int i=0; i<arrival.length; i++) {
            int time = arrival[i];
            while(!busy.isEmpty() && busy.peek()[0] <= time) {
                int server = busy.poll()[1];
                available.add(server);
            }
            if(available.isEmpty()) {
                continue;
            }
            int target = i % k;
            Integer targetServer = available.ceiling(target);
            if(targetServer == null) {
                targetServer = available.first();
            }
            available.remove(targetServer);
            busy.offer(new int[]{time + load[i], targetServer});
            countReq[targetServer]++;
        }

        int maxReq = 0;
        for(int c : countReq) {
            maxReq = Math.max(maxReq, c);
        }

        List<Integer> result = new ArrayList<>();
        for(int i=0; i<k; i++) {
            if(countReq[i] == maxReq) {
                result.add(i);
            }
        }

        return result;
    }
}
