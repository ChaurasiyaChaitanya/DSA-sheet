class Solution {
    public int[] assignTasks(int[] servers, int[] tasks) {
        int m = servers.length;
        int n = tasks.length;

        PriorityQueue<int[]> busy = new PriorityQueue<>((a, b) -> a[0] - b[0]);

        PriorityQueue<int[]> idle = new PriorityQueue<>((a, b) -> {
            if(a[0] != b[0]) {
                return a[0] - b[0];
            }
            return a[1] - b[1];
        });

        for(int i=0; i<m; i++) {
            idle.offer(new int[]{servers[i], i});
        }

        int[] result = new int[n];
        int ts = 0;
        for(int i=0; i<n; i++) {
            ts = Math.max(ts, i);
            release(busy, idle, servers, ts);
            if(idle.isEmpty()) {
                ts = busy.peek()[0];
                release(busy, idle, servers, ts);
            }

            int[] server = idle.poll();
            int idx = server[1];
            result[i] = idx;
            busy.offer(new int[]{ts + tasks[i], idx});
        }

        return result;
    }

    private void release(PriorityQueue<int[]> busy, PriorityQueue<int[]> idle, int[] servers, int ts) {
        while(!busy.isEmpty() && busy.peek()[0] <= ts) {
            int[] server = busy.poll();
            int idx = server[1];
            idle.offer(new int[]{servers[idx], idx});
        }
    }
}
