class Solution {
    private int getDigitSum(int num) {
        int sum = 0;
        while(num > 0) {
            sum += num % 10;
            num /= 10;
        }
        return sum;
    }

    public int minSwaps(int[] nums) {
        int n = nums.length;

        Integer[] indices = new Integer[n];
        for(int i=0; i<n; i++) {
            indices[i] = nums[i];
        }

        Arrays.sort(indices, (a, b) -> {
            int sum1 = getDigitSum(a);
            int sum2 = getDigitSum(b);
            if(sum1 != sum2) {
                return Integer.compare(sum1, sum2);
            }
            return Integer.compare(a, b);
        });

        Map<Integer, Integer> valIdx = new HashMap<>();
        for(int i=0; i<n; i++) {
            valIdx.put(nums[i], i);
        }

        int result = 0;
        for(int i=0; i<n; i++) {
            int expected = indices[i], curr = nums[i];
            if(expected != curr) {
                int idx = valIdx.get(expected);
                nums[i] = expected;
                nums[idx] = curr;
                valIdx.put(expected, i);
                valIdx.put(curr, idx);
                result++;
            }
        }

        return result;
    }
}
