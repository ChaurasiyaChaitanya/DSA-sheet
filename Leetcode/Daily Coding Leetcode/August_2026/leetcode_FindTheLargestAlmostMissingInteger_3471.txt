class Solution {
    public int largestInteger(int[] nums, int k) {
        int n = nums.length;
        if(n == k) {
            int result = nums[0];
            for(int x : nums) {
                result = Math.max(result, x);
            }
            return result;
        }

        int[] freq = new int[51];
        for(int num : nums) {
            freq[num]++;
        }
        if(k == 1) {
            for(int i=50; i>=0; i--) {
                if(freq[i] == 1) {
                    return i;
                }
            }

            return -1;
        }

        int result = -1;
        if(freq[nums[0]] == 1) {
            result = Math.max(result, nums[0]);
        }

        if(freq[nums[n-1]] == 1) {
            result = Math.max(result, nums[n-1]);
        }

        return result;
    }
}
