class Solution {
    public int maximumLengthSubstring(String s) {
        int n = s.length();
        int[] freq = new int[26];
        int i = 0, j = 0;
        int result = 0;

        while(j < n) {
            int c = s.charAt(j) - 'a';
            freq[c]++;
            while(freq[c] > 2) {
                freq[s.charAt(i) - 'a']--;
                i++;
            }
            result = Math.max(result, j - i + 1);
            j++;
        }

        return result;
    }
}
