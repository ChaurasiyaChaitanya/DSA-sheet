class Solution {
    public String shortestBeautifulSubstring(String s, int k) {
        int n = s.length();
        int i = 0, j = 0;
        int ones = 0;
        String result = "";

        while(j < n) {
            if(s.charAt(j) == '1') {
                ones++;
            }
            while(i < n && (ones > k || s.charAt(i) == '0')) {
                if(s.charAt(i) == '1') {
                    ones--;
                }
                i++;
            }
            if(ones == k) {
                String str = s.substring(i, j+1);
                if(result.isEmpty() || j - i + 1 < result.length() 
                    || (j - i + 1 == result.length() && str.compareTo(result) < 0)) {
                        result = str;
                }
            }
            j++;
        }

        return result;
    }
}
