class Solution {
    public int countRotations(String s, int k) {
        int n = s.length();
        String str = s + s;
        int result = 0;

        for(int i=0; i<n; i++) {
            int count = 0;
            for(int j=i; j<n+i-1; j++) {
                if(str.charAt(j) == str.charAt(j+1)) {
                    count++;
                }
                if(count > k) break;
            }
            if(count == k) {
                result++;
            }
        }

        return result;
    }
}
