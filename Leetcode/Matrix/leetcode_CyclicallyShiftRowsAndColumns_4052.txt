class Solution {
    public int[][] cyclicShift(int n, int[][] grid, int[] rowShift, int[] colShift) {
        int[][] shifted = new int[n][n];

        for(int r=0; r<n; r++) {
            for(int c=0; c<n; c++) {
                shifted[r][(c - rowShift[r] + n) % n] = grid[r][c];
            }
        }

        int[][] result = new int[n][n];

        for(int c=0; c<n; c++) {
            for(int r=0; r<n; r++) {
                result[(r - colShift[c] + n) % n][c] = shifted[r][c];
            }
        }

        return result;
    }
}
